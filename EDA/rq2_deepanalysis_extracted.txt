# ── Pipeline 2: Yaş Rezidüalizasyonu ──
residuals_asd = np.zeros((len(df_asd), len(THICKNESS_COLS)))
r2_per_region = []

for i, col in enumerate(THICKNESS_COLS):
    model = LinearRegression().fit(age_ctrl, df_ctrl[col].values)
    residuals_asd[:, i] = df_asd[col].values - model.predict(age_asd)
    ss_res = np.sum((df_ctrl[col].values - model.predict(age_ctrl))**2)
    ss_tot = np.sum((df_ctrl[col].values - df_ctrl[col].mean())**2)
    r2_per_region.append(1 - ss_res / ss_tot)

r2_arr = np.array(r2_per_region)

rq1_idx = [THICKNESS_COLS.index(r) for r in RQ1_REGIONS]
X_res_68  = StandardScaler().fit_transform(residuals_asd)

### CELL 7 - TYPE: code
def run_clustering(X, label, k=2, n_boot=100):
    """K-Means uygular, metrikleri ve stabiliteyi hesaplar."""
    km = KMeans(n_clusters=k, n_init=30, random_state=SEED)
    labels = km.fit_predict(X)

    sil = silhouette_score(X, labels)
    ch  = calinski_harabasz_score(X, labels)
    db  = davies_bouldin_score(X, labels)

    # Bootstrap stabilite
    aris = []
    for b in range(n_boot):
        rng = np.random.RandomState(b)
        idx = rng.choice(len(X), size=int(0.8*len(X)), replace=False)
        l_b = KMeans(n_clusters=k, n_init=10, random_state=b).fit_predict(X[idx])
        aris.append(adjusted_rand_score(labels[idx], l_b))

    # Yaş korelasyonu
    r_age, p_age = stats.pearsonr(labels, df_asd['AGE_AT_SCAN'].values)

    # Küme boyutları
    sizes = [(labels==c).sum() for c in range(k)]

    return {
        'label': label, 'labels': labels,
        'Silhouette': sil, 'CH': ch, 'DB': db,
        'ARI_mean': np.mean(aris), 'ARI_std': np.std(aris),
        'r_age': r_age, 'p_age': p_age,
        'sizes': sizes, 'X': X
    }

configs = {
    'Ham · 68 bölge':    run_clustering(X_ham_68,  'Ham · 68 bölge'),
    'Ham · 8 bölge':     run_clustering(X_ham_8,   'Ham · 8 bölge'),
    'Resid · 68 bölge':  run_clustering(X_res_68,  'Resid · 68 bölge'),
    'Resid · 8 bölge':   run_clustering(X_res_8,   'Resid · 8 bölge'),

## Blok 4 — PCA Loading Analizi

**PC1 loadings:** Hangi bölgeler PC1'e (ayrışma eksenine) en çok katkı yapıyor?

**Hipotez 1 (Global Kalınlık):** Tüm bölgelerin loadings değerleri aynı yönde ve birbirine yakın  
→ PC1 sadece "genel olarak kalın mı, ince mi?" sorusunu yanıtlıyor

**Hipotez 2 (Bölgesel Alt Tip):** Bazı bölgeler güçlü pozitif, bazıları güçlü negatif  
→ PC1 bir kalınlık *kontrastını* temsil ediyor — belirli bölgeler kalın, diğerleri ince

**4 konfigürasyon için ayrı ayrı hesaplanıyor.**


==================================================

### CELL 10 - TYPE: code
def pca_loading_analysis(X, feature_names, config_name):
    """PCA loading analizi — H1 vs H2 testi için."""
    pca = PCA(n_components=min(10, len(feature_names)), random_state=SEED)
    pca.fit(X)

    # PC1 loadings
    pc1_load = pca.components_[0]
    pc2_load = pca.components_[1]

    # H1 test: loadings ne kadar aynı yönde?
    # Eğer H1 doğruysa, hepsi pozitif veya hepsi negatif olmalı
    same_sign_frac = max(
        (pc1_load > 0).sum() / len(pc1_load),
        (pc1_load < 0).sum() / len(pc1_load)
    )
    # Loadings std: düşük std = hepsi benzer büyüklükte = H1
    load_std = pc1_load.std()
    load_range = pc1_load.max() - pc1_load.min()

    # En güçlü pozitif ve negatif katkıcılar
    top_pos = np.argsort(pc1_load)[::-1][:5]
    top_neg = np.argsort(pc1_load)[:5]

    return {
        'pca': pca,
        'pc1_load': pc1_load,
        'pc2_load': pc2_load,
        'feature_names': feature_names,
        'same_sign_frac': same_sign_frac,
        'load_std': load_std,
        'load_range': load_range,
        'top_pos': top_pos,
        'top_neg': top_neg,
        'var_exp': pca.explained_variance_ratio_,
        'config_name': config_name
    }

# 68 bölge için feature isimleri
feat_68 = THICKNESS_COLS
feat_8  = RQ1_REGIONS

loading_results = {
    'Ham · 68 bölge':   pca_loading_analysis(X_ham_68,  feat_68, 'Ham · 68 bölge'),
    'Ham · 8 bölge':    pca_loading_analysis(X_ham_8,   feat_8,  'Ham · 8 bölge'),
    'Resid · 68 bölge': pca_loading_analysis(X_res_68,  feat_68, 'Resid · 68 bölge'),
    'Resid · 8 bölge':  pca_loading_analysis(X_res_8,   feat_8,  'Resid · 8 bölge'),
}

### CELL 13 - TYPE: markdown
## Blok 5 — H1 vs H2 Hipotez Testi

**H1 (Global Kalınlık Farkı):** İki küme arasındaki fark, genel korteks kalınlığından kaynaklanıyor.  
Test: Küme 0 ve Küme 1'in *ortalama korteks kalınlığı* anlamlı farklı mı?

**H2 (Bölgesel Alt Tip):** İki küme, belirli bölgelerde zıt yönde farklılık gösteriyor.  
Test: Bazı bölgeler Küme 0'da daha kalın iken Küme 1'de daha ince mi?

**Dissociation indeksi:** Yüksekse H2, düşükse H1.


==================================================

### CELL 14 - TYPE: code
# En iyi konfigürasyon: Resid 8 bölge

print("H1 vs H2 HİPOTEZ TESTİ")
print("Konfigürasyon: Rezidüalize · 8 RQ1 Bölgesi · K-Means k=2")
print("="*65)

# H1 Testi: Ortalama korteks kalınlığı farkı
mean_thick_c0 = df_asd_clust[df_asd_clust['Cluster']==0][THICKNESS_COLS].mean(axis=1)
mean_thick_c1 = df_asd_clust[df_asd_clust['Cluster']==1][THICKNESS_COLS].mean(axis=1)
t_h1, p_h1 = stats.ttest_ind(mean_thick_c0, mean_thick_c1)
d_h1 = (mean_thick_c0.mean() - mean_thick_c1.mean()) / np.sqrt(
    (mean_thick_c0.std()**2 + mean_thick_c1.std()**2) / 2)

region_diffs = []
dissociation_pairs = []
for col in RQ1_REGIONS:
    v0 = df_asd_clust[df_asd_clust['Cluster']==0][col]
    v1 = df_asd_clust[df_asd_clust['Cluster']==1][col]
    t, p = stats.ttest_ind(v0, v1)
    d = (v0.mean() - v1.mean()) / np.sqrt((v0.std()**2 + v1.std()**2) / 2)
    region_diffs.append({'Region': col, 't': t, 'p': p, 'd': d,
                         'mean_c0': v0.mean(), 'mean_c1': v1.mean()})

df_rdiff = pd.DataFrame(region_diffs).sort_values('d', ascending=False)
print(f"\n{'Bölge':35s} {'d(C0-C1)':>10} {'p':>10} {'C0 orta':>10} {'C1 orta':>10}")
print("─"*75)
for _, row in df_rdiff.iterrows():
    sig = '*' if row['p'] < 0.05 else ''
    direction = '↑C0' if row['d'] > 0 else '↓C0'
    print(f"{row['Region']:35s} {row['d']:>10.4f} {row['p']:>10.4f}  "
          f"{row['mean_c0']:>8.4f}  {row['mean_c1']:>8.4f}  {direction}{sig}")

# Dissociation indeksi: bazı bölgeler + bazıları - ise yüksek
signs = np.sign(df_rdiff['d'].values)
dissociation_idx = 1 - abs(signs.mean())  # 0=hepsi aynı yön, 1=tam karışık
print(f"\nDissociation indeksi: {dissociation_idx:.4f}  (0=global, 1=tam bölgesel)")
h2_verdict = "H2 DESTEKLENIYOR ✅" if dissociation_idx > 0.3 else "H2 DESTEKLENMİYOR ✗"
print(f"→ {h2_verdict}")

### CELL 17 - TYPE: code
# LDA — 68 bölge ve 8 bölge için
results_lda = {}

for feat_name, X, fnames in [
    ('Ham · 68 bölge',   X_ham_68,  THICKNESS_COLS),
    ('Ham · 8 bölge',    X_ham_8,   RQ1_REGIONS),
    ('Resid · 68 bölge', X_res_68,  THICKNESS_COLS),
    ('Resid · 8 bölge',  X_res_8,   RQ1_REGIONS),
]:
    lbl = configs[feat_name]['labels']
    lda = LDA(n_components=1)
    lda.fit(X, lbl)
    lda_scores = lda.transform(X).ravel()
    coefs = lda.coef_[0]

    # LDA ayrışma kalitesi
    t, p = stats.ttest_ind(lda_scores[lbl==0], lda_scores[lbl==1])
    d = (lda_scores[lbl==0].mean() - lda_scores[lbl==1].mean()) / np.sqrt(
        (lda_scores[lbl==0].std()**2 + lda_scores[lbl==1].std()**2) / 2)

    results_lda[feat_name] = {
        'coefs': coefs, 'fnames': fnames,
        'lda_scores': lda_scores, 'labels': lbl,
        't': t, 'p': p, 'd': d
    }

print("LDA Ayrışma Kalitesi:")
print(f"{'Konfigürasyon':22s} {'t-stat':>10} {'p-değeri':>12} {'Cohen d':>10}")
print("─"*58)
for name, res in results_lda.items():
    print(f"{name:22s} {res['t']:>10.3f} {res['p']:>12.6f} {res['d']:>10.4f}")

## Blok 8 — Pipeline Etkisi Ölçümü

Harmonizasyon ve yaş düzeltmesinin kümeleme kalitesine katkısını ölçüyoruz.

**Ölçüm kriterleri:**
- Silhouette (cluster kalitesi ↑)
- Bootstrap ARI (stabilite ↑)
- r(küme, yaş) (yaş bağımlılığı ↓ → 0'a yakın iyi)
- r(küme, site) (site bağımlılığı ↓ → 0'a yakın iyi)


==================================================

### CELL 24 - TYPE: code
# Her konfigürasyon için site korelasyonu da hesapla
from sklearn.preprocessing import LabelEncoder

le_site = LabelEncoder()
site_encoded = le_site.fit_transform(df_asd['SITE_ID'].values)

pipeline_summary = {}
for name, res in configs.items():
    lbl = res['labels']
    r_age,  p_age  = stats.pearsonr(lbl, df_asd['AGE_AT_SCAN'].values)
    r_site, p_site = stats.pointbiserialr(lbl, site_encoded)
    pipeline_summary[name] = {
        'Silhouette':  res['Silhouette'],
        'ARI':         res['ARI_mean'],
        '|r_age|':     abs(r_age),
        '|r_site|':    abs(r_site),
        'p_age':       p_age,
        'p_site':      p_site,
    }

print("PİPELİNE ETKİSİ ÖZET TABLOSU")
print("="*80)
print(f"{'Konfigürasyon':22s} {'Sil↑':>8} {'ARI↑':>8} {'|r_age|↓':>10} {'|r_site|↓':>11}")
print("─"*65)
for name, res in pipeline_summary.items():
    age_ok  = '✅' if res['|r_age|']  < 0.15 else '⚠️'
    site_ok = '✅' if res['|r_site|'] < 0.10 else '⚠️'
    print(f"{name:22s} {res['Silhouette']:>8.4f} {res['ARI']:>8.4f} "
          f"{res['|r_age|']:>8.4f} {age_ok}  {res['|r_site|']:>9.4f} {site_ok}")

print()
print("Yorumlama:")
print("  |r_age|  < 0.15 → yaştan yeterince bağımsız ✅")
print("  |r_site| < 0.10 → site etkisinden yeterince bağımsız ✅")
